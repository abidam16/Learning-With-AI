package main

import (
	"fmt"
	"os"
	"strings"
)

func main() {
	fmt.Println("Impor: mulai")
	data, err := os.ReadFile("/data/orders.txt")
	if err != nil {
		fmt.Fprintln(os.Stderr, "Detail:", err)
		fmt.Println("Impor: gagal")
		os.Exit(1)
	}
	rows := strings.Fields(string(data))
	fmt.Printf("Impor: selesai, %d pesanan\n", len(rows))
}
